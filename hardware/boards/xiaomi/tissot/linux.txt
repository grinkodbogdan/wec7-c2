# The Linux driver host on this board (docs/linux-drivers.md).
#
# linux.dtb is the Mi A1's mainline device tree: postmarketOS's
# msm8953-mainline (Linux 6.12) msm8953-xiaomi-tissot.dts, compiled, as QRT
# ships it (src/arch/arm64/dts/; its Makefile's dts-tissot target rebuilds it).
# The composition gives Linux the whole tree ([linux] devicetree = true):
# every device below that is not dropped is Linux's, with its registers and
# interrupts, as on a Linux phone - except what Flare's own drivers hold,
# which the build leaves out. QRT's notes on this phone (README 0.21-0.28)
# say which devices must not run: they are dropped here.
#
#   drop PATH      Linux never sees it (with everything under it)
#   hide PATH      Linux sees it only once the driver host has its firmware
#                  ready: one stage per line, in order (`linuxdrv`)
#   delprop/setprop/cmdline  as below

# ---- Flare's own ----
# The CPUs: their timers, PMU, power states, clock (APCS PLL), voltage (CPR)
# and thermal limits.
drop /timer
drop /pmu
drop /psci
drop /thermal-zones
drop /soc@0/timer@b120000
drop /soc@0/clock-controller@b011050
drop /soc@0/power-controller@b018000
# The panel: Flare's display driver (drivers/qcom/mdp5-display) scans out of
# it; its IOMMU (the display's and video's) must keep passing the panel's
# reads through; the debug UART is Flare's console pads.
drop /soc@0/display-subsystem@1a00000
drop /soc@0/iommu@1e20000
drop /soc@0/serial@78af000
# The bus fabric (NoCs): the RPM keeps the boot loader's bus clocks.
drop /soc@0/interconnect@400000
drop /soc@0/interconnect@500000
drop /soc@0/interconnect@580000

# ---- What can reset or power off the phone by itself (QRT 0.21.14) ----
# PS_HOLD (Linux restarts the phone through it), the thermal sensors and the
# PMIC's temperature alarm (a misread "critical" powers it off), the charger
# and the display's bias rails (LAB/IBB) reprogram the power hardware.
drop /soc@0/restart@4ab000
drop /soc@0/thermal-sensor@4a9000
drop /soc@0/spmi@200f000/pmic@0/temp-alarm@2400
drop /soc@0/spmi@200f000/pmic@2/smbchager@1000
drop /soc@0/spmi@200f000/pmic@3/labibb

# ---- Not yet: the modem, the camera, video and the IP accelerator ----
drop /soc@0/remoteproc@4080000
drop /smp2p-modem
drop /smsm/modem@1
drop /memshare
drop /soc@0/camss@1b00020
drop /soc@0/cci@1b0c000
drop /soc@0/venus@1d00000
drop /soc@0/ipa@7900000

# ---- After the shell is up, one stage at a time ----
# At boot Linux gets what the phone needs to start: eMMC, touch, keys, USB,
# the RPM and clocks, the PMICs' RTC, power key and GPIOs. Everything else
# comes in stages (`linuxdrv`, once the shell is up), each announced on the
# console before it starts, so a device that hangs or resets the phone
# names itself: the stage printed last.
# 1. the PMICs' battery gauge, ADCs, backlight, vibrator and charge LED
hide /soc@0/spmi@200f000/pmic@2/fuel-gauge@4000 /soc@0/spmi@200f000/pmic@2/adc@3100 /soc@0/spmi@200f000/pmic@0/adc@3100 /soc@0/spmi@200f000/pmic@3/leds@d800 /soc@0/spmi@200f000/pmic@3/haptics@c000 /soc@0/spmi@200f000/pmic@2/leds@1243
# 2. the other buses and what is on them: sensors (bit-banged I2C), the
#    second I2C and SPI controllers and their DMA, the RNG, the fuses
hide /i2c-sensors /soc@0/i2c@78b6000 /soc@0/spi@7af6000 /soc@0/dma-controller@7ac4000 /soc@0/rng@e3000 /soc@0/efuse@a4000
# 3-7. through the secure world, once the driver host has their firmware:
# its call interface (SCM), the GPU's IOMMU, the GPU (Adreno 506; CP
# microcode and zap shader from the vendor partition), Wi-Fi (the Pronto
# core; its firmware from the modem partition), the audio DSP with the
# sound card, codecs and speaker amplifier.
hide /firmware
hide /soc@0/iommu@1c48000
hide /soc@0/gpu@1c00000
hide /soc@0/remoteproc@a204000 /smp2p-wcnss /smsm/wcnss@6
hide /soc@0/remoteproc@c200000 /smp2p-adsp /soc@0/sound-card@c051000 /soc@0/codec@c0f0000 /soc@0/spmi@200f000/pmic@1/codec@f000 /analog-amplifier

# Bus bandwidth (interconnect) paths: the NoCs are not Linux's (above).
# Without the references the SDHC does not wait for providers that never
# appear.
delprop interconnects
delprop interconnect-names
# ... and the bandwidths in operating-point tables, which Linux's OPP code
# rejects without paths (the SDHC's; rates and corners stay)
delprop opp-peak-kBps
delprop opp-avg-kBps

# USB in device mode only (fdb over the USB serial gadget): the controller's
# role would follow the charger's ID detection (extcon), which Linux does
# not get here.
setprop /soc@0/usb@70f8800/usb@7000000 dr_mode peripheral
delprop extcon

# Linux must not switch off what it does not know is in use: the panel's
# clocks and power domain and the UART's clock (Flare's console), and the
# rails the boot loader left on.
cmdline clk_ignore_unused pd_ignore_unused regulator_ignore_unused
# Probe order by the tree's references, but never wait forever for a
# supplier Linux does not have (as QRT boots it).
cmdline fw_devlink=permissive
# Every message to the driver host, which keeps them (and shows their tail
# if Linux has not brought its devices up after 30 s) but prints only the
# ones that look like trouble.
cmdline loglevel=7
